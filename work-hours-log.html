Google Sheets sync

New entries can auto-append to your Google Sheet via a small Apps Script you connect once. This only works once the page is hosted outside Claude (e.g. GitHub Pages) — see setup below.

Setup instructions
  1. Open your sheet → Extensions → Apps Script.
  2. Delete any placeholder code and paste the script below.
  3. Click Deploy → New deployment → type "Web app". Set "Execute as" to yourself and "Who has access" to Anyone.
  4. Deploy, authorize when prompted, then copy the Web app URL into the field above and click Save.
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  var data = JSON.parse(e.postData.contents);
  var rows = Array.isArray(data) ? data : [data];
  rows.forEach(function (r) {
    sheet.appendRow([r.date, r.dayOfWeek, r.time, r.hours, r.task, r.category]);
  });
  return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
    .setMimeType(ContentService.MimeType.JSON);
}
